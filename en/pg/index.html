// POST /api/payments/confirm  (Express)
import express from 'express';
const app = express();
app.use(express.json());

const SECRET_KEY = process.env.KT_SECRET_KEY; // test_sk_...
const auth = 'Basic ' + Buffer.from(SECRET_KEY + ':').toString('base64');

app.post('/api/payments/confirm', async (req, res) => {
  const { paymentKey, orderId, amount } = req.body;

  // Verify the amount — always compare with your DB
  const order = await db.orders.find(orderId);
  if (order.amount !== amount) return res.status(400).json({ code: 'AMOUNT_MISMATCH' });

  const r = await fetch('https://api.koratechit.com/v1/payments/confirm', {
    method: 'POST',
    headers: { Authorization: auth, 'Content-Type': 'application/json' },
    body: JSON.stringify({ paymentKey, orderId, amount })
  });
  const data = await r.json();
  res.status(r.status).json(data);
});

01Understand the payment flow

A KORATECH payment has three stages: request → authenticate → confirm. After the buyer authenticates, they're redirected to your successUrl; the payment completes only when your server calls the confirm API.

  1. RequestOpen checkout from the client with the SDK.
  2. AuthenticateBuyer completes card or wallet authentication.
  3. RedirectpaymentKey, orderId and amount are returned.
  4. ConfirmCall the confirm API — status becomes DONE.
💡If the confirm API isn't called within 10 minutes of authentication, the payment expires automatically (EXPIRED).

  document.getElementById('payBtn').addEventListener('click', () => {
    kt.requestPayment({
      method: 'CARD',            // CARD | EASY_PAY | TRANSFER | VIRTUAL_ACCOUNT

      orderName: 'Wool Blend Half Coat and 1 more',

  });

import { loadKoratechPay } from '@koratech/pay-sdk';

export default function CheckoutButton({ order }) {
  const onPay = async () => {
    const kt = await loadKoratechPay(process.env.NEXT_PUBLIC_KT_CLIENT_KEY);
    await kt.requestPayment({
      method: 'EASY_PAY',
      amount: order.amount,
      orderId: order.id,
      orderName: order.name,
      successUrl: `${window.location.origin}/success`,
      failUrl: `${window.location.origin}/fail`
    });
  };
  return <button onClick={onPay}>Pay now</button>;
}

05Webhooks

Async events such as virtual account deposits and cancellations arrive via webhook. Verify the KT-Signature header on every request.

import crypto from 'crypto';

app.post('/webhooks/koratech', express.raw({ type: 'application/json' }), (req, res) => {
  const sig = req.header('KT-Signature');
  const expected = crypto.createHmac('sha256', process.env.KT_WEBHOOK_SECRET)
    .update(req.body).digest('hex');
  if (sig !== expected) return res.sendStatus(401);

  const event = JSON.parse(req.body);
  if (event.eventType === 'PAYMENT_STATUS_CHANGED') {
    // event.data.status: DONE | CANCELED | WAITING_FOR_DEPOSIT ...
  }
  res.sendStatus(200); // respond 200 within 10s; retried up to 7 times on failure
});
